CMAKE_MINIMUM_REQUIRED( VERSION 3.10 )

PROJECT(${CONVO_NETWORKING_LIB_NAME})

FILE(GLOB SOURCES *.cpp)

IF (NOT CONVO_NO_BUILD_WORKQUEUE)
  MESSAGE("building sub:networking:workqueue")
  FILE(GLOB ADAPTER_SOURCES WorkQueue/*.cpp)
  LIST(APPEND SOURCES ${ADAPTER_SOURCES})

  IF (NOT CONVO_NO_BUILD_MSGBUS)
     MESSAGE("building sub:networking:msgbus")
     FILE(GLOB MSGBUS_SOURCES MessageBus/*.cpp)
     LIST(APPEND SOURCES ${MSGBUS_SOURCES})
  ENDIF ()
ENDIF ()


FILE(GLOB HEADERS *.h)

ADD_LIBRARY( ${CONVO_NETWORKING_LIB_NAME} ${SOURCES} ${HEADERS})

TARGET_INCLUDE_DIRECTORIES(${CONVO_NETWORKING_LIB_NAME} PRIVATE )

TARGET_LINK_LIBRARIES(${CONVO_NETWORKING_LIB_NAME}
   ${CONVO_BASE_LIB_NAME}
)
